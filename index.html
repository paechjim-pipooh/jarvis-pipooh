// โหลดรายการเสียงและเลือกเสียงภาษาไทยอัตโนมัติทันที let voices = []; let selectedVoiceIndex = 0; function loadVoices() { voices = window.speechSynthesis.getVoices(); const voiceSelect = document.getElementById('voiceSelect'); if (voices.length > 0) { // ค้นหาเสียงภาษาไทยตัวแรกในระบบ (หรือเสียง Siri ไทย ถ้ามี) let thaiIndex = voices.findIndex(v => v.lang.includes('th') || v.lang.includes('th-TH')); if (thaiIndex !== -1) { selectedVoiceIndex = thaiIndex; // ล็อคเป้าใช้เสียงไทยทันที } else { // ถ้าหาไม่เจอ ให้ใช้เสียงสำเนียงอังกฤษมาตรฐาน let enIndex = voices.findIndex(v => v.lang.includes('en-US')); if (enIndex !== -1) selectedVoiceIndex = enIndex; } // อัปเดตรายการในหน้า Settings เผื่อต้องการเปลี่ยน if (voiceSelect) { voiceSelect.innerHTML = ''; voices.forEach((voice, i) => { if (voice.lang.includes('th') || voice.lang.includes('en')) { const option = document.createElement('option'); option.value = i; option.textContent = `${voice.name} (${voice.lang})`; if (i === selectedVoiceIndex) option.selected = true; voiceSelect.appendChild(option); } }); } } } if (typeof window.speechSynthesis !== 'undefined') { window.speechSynthesis.onvoiceschanged = loadVoices; loadVoices(); } // ฟังก์ชันเปลี่ยนเสียงเมื่อผู้ใช้เลือกเองใน Settings function changeVoice() { const voiceSelect = document.getElementById('voiceSelect'); if (voiceSelect) { selectedVoiceIndex = voiceSelect.value; speakText("เปลี่ยนเสียงเรียบร้อยครับ คุณพิภู"); } }
